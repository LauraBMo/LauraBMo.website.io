{{- /* Render LaTeX at build time using Hugo's embedded KaTeX. Nothing ships to the browser but static markup — no scripts, no runtime work, no flash of raw `$…$`. output = "htmlAndMathml" emits KaTeX's positioned HTML for sighted readers plus a parallel MathML tree for assistive tech (the visual half carries aria-hidden, so screen readers only ever see the MathML). The HTML half is meaningless without katex.css, which is the one exemption in the "legible with CSS disabled" rule. hasMath tells layouts/_partials/head-katex.html to link that stylesheet. */ -}} {{- .Page.Store.Set "hasMath" true -}} {{- $opts := dict "output" "htmlAndMathml" "displayMode" (eq .Type "block") -}} {{- with try (transform.ToMath .Inner $opts) -}} {{- with .Err -}} {{- errorf "KaTeX: %s\n in %s\n source: %s" . $.Page.File.Filename $.Inner -}} {{- else -}} {{- .Value -}} {{- end -}} {{- end -}}
